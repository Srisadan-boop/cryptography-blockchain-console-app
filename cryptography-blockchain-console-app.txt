import hashlib
import sys
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.asymmetric import rsa, padding
from cryptography.hazmat.primitives import serialization
from cryptography.hazmat.backends import default_backend
from cryptography.exceptions import InvalidSignature


# ----------------------------
# Vehicle Registration Storage
# ----------------------------
vehicle_db = {}

# ----------------------------
# Generate RSA Key Pair (once)
# ----------------------------
private_key = rsa.generate_private_key(
    public_exponent=65537,
    key_size=2048,
    backend=default_backend()
)

public_key = private_key.public_key()


# ----------------------------
# SHA-256 Hash Function
# ----------------------------
def generate_sha256():
    message = input("Enter message to hash: ").strip()
    
    if not message:
        print("Message cannot be empty.")
        return
    
    sha256_hash = hashlib.sha256(message.encode()).hexdigest()
    print(f"SHA-256 Hash:\n{sha256_hash}")


# ----------------------------
# Digital Signature Functions
# ----------------------------
def sign_message():
    message = input("Enter message to sign: ").strip()
    
    if not message:
        print("Message cannot be empty.")
        return
    
    signature = private_key.sign(
        message.encode(),
        padding.PSS(
            mgf=padding.MGF1(hashes.SHA256()),
            salt_length=padding.PSS.MAX_LENGTH
        ),
        hashes.SHA256()
    )

    print("Message signed successfully.")
    print(f"Signature (hex):\n{signature.hex()}")

    return message, signature


def verify_signature():
    message = input("Enter original message: ").strip()
    signature_hex = input("Enter signature (hex): ").strip()

    try:
        signature = bytes.fromhex(signature_hex)

        public_key.verify(
            signature,
            message.encode(),
            padding.PSS(
                mgf=padding.MGF1(hashes.SHA256()),
                salt_length=padding.PSS.MAX_LENGTH
            ),
            hashes.SHA256()
        )

        print("Signature is VALID ✅")

    except (InvalidSignature, ValueError):
        print("Signature is INVALID ❌")


# ----------------------------
# Vehicle Registration System
# ----------------------------
def register_vehicle():
    number_plate = input("Enter Number Plate: ").strip().upper()
    owner = input("Enter Owner Name: ").strip()
    model = input("Enter Vehicle Model: ").strip()

    if not number_plate or not owner or not model:
        print("All fields are required.")
        return

    if number_plate in vehicle_db:
        print("Error: Vehicle with this Number Plate already exists.")
        return

    vehicle_db[number_plate] = {
        "owner": owner,
        "model": model
    }

    print("Vehicle registered successfully.")


def retrieve_vehicle():
    number_plate = input("Enter Number Plate to search: ").strip().upper()

    if number_plate in vehicle_db:
        vehicle = vehicle_db[number_plate]
        print("Vehicle Found:")
        print(f"Owner: {vehicle['owner']}")
        print(f"Model: {vehicle['model']}")
    else:
        print("Vehicle not found.")


# ----------------------------
# Main Menu Loop
# ----------------------------
def main():
    while True:
        print("\n===== Cryptography & Blockchain Console App =====")
        print("1. Generate SHA-256 Hash")
        print("2. Sign a Message (Digital Signature)")
        print("3. Verify Digital Signature")
        print("4. Register Vehicle")
        print("5. Retrieve Vehicle")
        print("6. Exit")

        choice = input("Enter your choice (1-6): ").strip()

        if choice == "1":
            generate_sha256()
        elif choice == "2":
            sign_message()
        elif choice == "3":
            verify_signature()
        elif choice == "4":
            register_vehicle()
        elif choice == "5":
            retrieve_vehicle()
        elif choice == "6":
            print("Exiting program.")
            sys.exit()
        else:
            print("Invalid choice. Please enter 1-6.")


if __name__ == "__main__":
    main()
